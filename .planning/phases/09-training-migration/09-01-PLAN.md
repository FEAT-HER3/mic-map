      #ifdef _WIN32
          const auto tmp = std::filesystem::path(dest.wstring() + L".tmp");
          // 1) Ask detector to serialize binary profile to tmp via existing v1.5 ofstream path
          //    (noise_detector.cpp:329). Preserves binary format byte-for-byte.
          if (!detector.saveTrainingData(tmp)) {
              MICMAP_LOG_ERROR("training_io: detector.saveTrainingData(tmp) failed for ", tmp.string());
              return false;
          }
          // 2) ReplaceFileW dest with tmp (atomic on NTFS).
          if (std::filesystem::exists(dest)) {
              if (!ReplaceFileW(dest.c_str(), tmp.c_str(), nullptr,
                                REPLACEFILE_WRITE_THROUGH | REPLACEFILE_IGNORE_MERGE_ERRORS,
                                nullptr, nullptr)) {
                  const auto err = ::GetLastError();
                  MICMAP_LOG_ERROR("training_io: ReplaceFileW failed (GetLastError=", err, ")");

                  if (!MoveFileExW(tmp.c_str(), dest.c_str(),

                      MICMAP_LOG_ERROR("training_io: MoveFileExW fallback also failed");
                      std::error_code ec;
                      std::filesystem::remove(tmp, ec);
                      return false;

              // Fresh write — no atomic swap needed; just rename.
              if (!MoveFileExW(tmp.c_str(), dest.c_str(),

                  MICMAP_LOG_ERROR("training_io: MoveFileExW (fresh) failed");
                  std::error_code ec;
                  std::filesystem::remove(tmp, ec);
                  return false;
              }
          }
          backupAndRotate(dest);
          return true;
      #else
          (void)dest; (void)detector;
          return false;
      #endif
      }
