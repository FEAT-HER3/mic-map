**1. apps/mic_test/src/wav_replay.hpp** — header. Structure:

```cpp
/**
 * @file wav_replay.hpp
 * @brief Phase 9 / TEST-04 / D-28..D-37 — WAV replay harness for mic_test.
 *
 * Decode a WAV file (or directory of WAVs), feed samples through
 * INoiseDetector + IStateMachine deterministically, emit per-file results
 * in JSON shape per CONTEXT D-30. Bit-depth policy per D-32; pacing policy
 * flat-out per D-33; determinism per D-34.
 *
 * TEST-01 invariant: ZERO OpenVR symbols. AssertReplayNoVrApi enforces.
 */
#pragma once

#include <chrono>
#include <cstdint>
#include <filesystem>
#include <optional>
#include <string>
#include <vector>

namespace micmap::detection { class INoiseDetector; class IStateMachine; }

namespace micmap::mic_test {

struct ReplayConfig {
    std::filesystem::path config_path;       // for JSON output reporting
    std::filesystem::path profile_path;      // for JSON output reporting
    uint32_t target_sample_rate{48000};      // detector's expected rate
    size_t fft_size{1024};                   // detector's fft size
    int max_duration_s{600};                 // D-32: reject longer WAVs
    bool quiet{false};                       // suppress per-file stdout summary lines
    bool verbose{false};                     // emit per-trigger stdout lines
};

struct TriggerEvent {
    double t_s{0.0};                         // seconds from start of WAV
    float confidence{0.0f};
    std::string state;                       // "Triggered"
};

struct ReplayResult {
    int exit_code{0};                        // 0=ok / no expectations, 1=expectation failed, 2=format/IO error
    std::filesystem::path wav;
    double duration_s{0.0};
    uint32_t sample_rate{0};
    uint16_t channels{0};
    std::optional<int> expected_triggers;
    int observed_triggers{0};
    int tolerance{0};
    bool pass{true};
    std::vector<TriggerEvent> triggers;
    std::optional<std::string> error_message;
};

/// Replay a single WAV against a detector + state machine.
/// detector and sm must be pre-loaded (caller's responsibility — including INoiseDetector::loadTrainingData).
/// Returns ReplayResult with exit_code 0 on success-or-no-expectations,
/// 1 on expectation failure, 2 on IO/format error.
ReplayResult replayWav(const std::filesystem::path& wav,
                       const ReplayConfig& cfg,
                       micmap::detection::INoiseDetector& detector,
                       micmap::detection::IStateMachine& sm,
                       std::optional<int> expected_triggers = std::nullopt,
                       int tolerance = 0);

/// Replay a directory of WAVs (recursive). Returns vector of per-file results.
/// If expectations_path is provided, loads expected_triggers + tolerance per file from manifest.json.
```

```cpp
 * @file wav_replay.cpp
 * @brief Implementation of WAV replay harness. See wav_replay.hpp for API surface.
 *
 * D-32: stereo→mono averaged downmix; linear-interp sample-rate resample.
 *       16-bit PCM and 32-bit float supported; reject 8/24-bit with exit 2.

 * D-34: determinism — dt = frame_count * 1000 / sample_rate (NOT steady_clock).
 * D-37: ZERO OpenVR symbols. AssertReplayNoVrApi enforces.

constexpr size_t kBlockFrames = 480;   // matches driver SampleRing block size

// Convert interleaved L+R stereo to mono by averaging.
void downmixStereoToMono(const float* in_lr, size_t frame_count, std::vector<float>& out_mono) {
    out_mono.resize(frame_count);
    for (size_t i = 0; i < frame_count; ++i) {
        out_mono[i] = 0.5f * (in_lr[2 * i] + in_lr[2 * i + 1]);
    }
}

// Linear-interp resample mono buffer from src_rate to dst_rate.
// D-32: deterministic, low-quality (regression testing only — not for audio fidelity).
std::vector<float> linearResample(const std::vector<float>& in,
                                  uint32_t src_rate, uint32_t dst_rate) {
    if (src_rate == dst_rate) return in;
    const double ratio = static_cast<double>(dst_rate) / src_rate;
    const size_t out_count = static_cast<size_t>(in.size() * ratio + 0.5);
    std::vector<float> out(out_count);
    for (size_t i = 0; i < out_count; ++i) {
        const double src_pos = i / ratio;
        const size_t lo = static_cast<size_t>(src_pos);
        const size_t hi = std::min(lo + 1, in.size() - 1);
        const float frac = static_cast<float>(src_pos - lo);
        out[i] = in[lo] * (1.0f - frac) + in[hi] * frac;
    }
    return out;
}

ReplayResult makeError(const std::filesystem::path& wav, int code, const std::string& msg) {
    ReplayResult r;
    r.wav = wav;
    r.exit_code = code;
    r.error_message = msg;
    r.pass = false;
    std::cerr << "error: " << wav.string() << ": " << msg << "\n";
    return r;
}

ReplayResult replayWav(const std::filesystem::path& wav,
                       const ReplayConfig& cfg,
                       micmap::detection::INoiseDetector& detector,
                       micmap::detection::IStateMachine& sm,
                       std::optional<int> expected_triggers,
                       int tolerance) {
    if (!std::filesystem::exists(wav)) {
        return makeError(wav, 2, "file not found");
    }

    drwav w;
    if (!drwav_init_file_w(&w, wav.wstring().c_str(), nullptr)) {
        return makeError(wav, 2, "WAV open failed (drwav_init_file_w)");
    }

    // D-32: bit-depth gate.
    const bool is_16bit_pcm = (w.bitsPerSample == 16);
    const bool is_32bit_float = (w.bitsPerSample == 32 && w.translatedFormatTag == DR_WAVE_FORMAT_IEEE_FLOAT);
    if (!is_16bit_pcm && !is_32bit_float) {
        std::ostringstream msg;
        msg << "unsupported bit depth " << w.bitsPerSample
            << " (only 16-bit PCM and 32-bit float supported)";
        drwav_uninit(&w);
        return makeError(wav, 2, msg.str());
    }

    const double duration_s = static_cast<double>(w.totalPCMFrameCount) / w.sampleRate;
    if (duration_s > cfg.max_duration_s) {
        std::ostringstream msg;
        msg << "duration " << duration_s << "s exceeds --max-duration "
            << cfg.max_duration_s << "s";
        drwav_uninit(&w);
        return makeError(wav, 2, msg.str());
    }

    // Read entire file as f32 (interleaved if stereo).
    std::vector<float> raw(w.totalPCMFrameCount * w.channels);
    drwav_uint64 frames_read = drwav_read_pcm_frames_f32(&w, w.totalPCMFrameCount, raw.data());
    drwav_uninit(&w);
    if (frames_read != w.totalPCMFrameCount) {
        return makeError(wav, 2, "drwav_read_pcm_frames_f32 short read");
    }

    // Downmix stereo→mono if needed.
    std::vector<float> mono;
    if (w.channels == 1) {
        mono = std::move(raw);
    } else if (w.channels == 2) {
        downmixStereoToMono(raw.data(), w.totalPCMFrameCount, mono);
    } else {
        // Average all channels (handle 5.1, etc.).
        mono.resize(w.totalPCMFrameCount);
        for (size_t i = 0; i < w.totalPCMFrameCount; ++i) {
            float sum = 0.0f;
            for (uint16_t c = 0; c < w.channels; ++c) {
                sum += raw[i * w.channels + c];
            }
            mono[i] = sum / w.channels;
        }
    }

    // Resample to target rate if needed.
    const std::vector<float> samples = linearResample(mono, w.sampleRate, cfg.target_sample_rate);

    // Build result.
    ReplayResult r;
    r.wav = wav;
    r.duration_s = duration_s;
    r.sample_rate = w.sampleRate;
    r.channels = w.channels;
    r.expected_triggers = expected_triggers;
    r.tolerance = tolerance;

    // Run blocks through detector + state machine. D-33 flat-out pacing; D-34 dt from frame count.
    const uint32_t rate = cfg.target_sample_rate;
    const uint32_t rate_for_dt = rate ? rate : 1;
    micmap::detection::DetectionState last_state = micmap::detection::DetectionState::Idle;

    size_t processed = 0;
    while (processed < samples.size()) {
        const size_t this_block = std::min<size_t>(kBlockFrames, samples.size() - processed);
        auto result = detector.analyze(samples.data() + processed, this_block);
        const auto dt = std::chrono::milliseconds(
            static_cast<long long>(this_block) * 1000 / rate_for_dt);
        sm.update(result.confidence, dt);

        const auto state = sm.getState();
        if (state == micmap::detection::DetectionState::Triggered
                && last_state != micmap::detection::DetectionState::Triggered) {
            // Rising edge — record a trigger.
            TriggerEvent ev;
            ev.t_s = static_cast<double>(processed) / rate;
            ev.confidence = result.confidence;
            ev.state = "Triggered";
            r.triggers.push_back(ev);
            r.observed_triggers++;
            if (cfg.verbose) {
                std::cout << "TRIGGER  " << wav.string() << "  t=" << ev.t_s
                          << "s  confidence=" << ev.confidence << "\n";
            }
        }
        last_state = state;
        processed += this_block;
    }

    // Evaluate against expectation.
    if (r.expected_triggers.has_value()) {
        const int observed = r.observed_triggers;
        const int expected = *r.expected_triggers;
        const int diff = std::abs(observed - expected);
        r.pass = diff <= tolerance;
        r.exit_code = r.pass ? 0 : 1;
    } else {
        r.pass = true;     // no expectations = always pass
        r.exit_code = 0;
    }

    if (!cfg.quiet) {
        std::cout << (r.pass ? "PASS" : "FAIL") << "  " << wav.string()
                  << "  observed=" << r.observed_triggers
                  << " expected=" << (r.expected_triggers.has_value()
                        ? std::to_string(*r.expected_triggers) : "n/a")
                  << " (tolerance ±" << r.tolerance << ")\n";
    }
    return r;
}

DirReplayResult replayWavDir(const std::filesystem::path& dir,
                             const ReplayConfig& cfg,
                             micmap::detection::INoiseDetector& detector,
                             micmap::detection::IStateMachine& sm,
                             const std::filesystem::path& expectations_path) {
    DirReplayResult result;

    // Load manifest if provided.
    std::map<std::string, std::pair<int, int>> expectations;   // filename → (expected, tolerance)
    if (!expectations_path.empty() && std::filesystem::exists(expectations_path)) {
        try {
            std::ifstream f(expectations_path);
            nlohmann::json m = nlohmann::json::parse(f);
            const auto& files_array = m.is_array() ? m : m.value("files", nlohmann::json::array());
            for (const auto& entry : files_array) {
                const auto name = entry.value("wav", std::string{});
                const auto exp = entry.value("expected_triggers", 0);
                const auto tol = entry.value("tolerance", 0);
                if (!name.empty()) {
                    expectations[name] = {exp, tol};
                }
            }
        } catch (const std::exception& e) {
            std::cerr << "manifest parse failed: " << e.what() << "\n";
        }
    }

    // Collect WAV files (recursive, sorted for determinism per D-34).
    std::vector<std::filesystem::path> wavs;
    for (auto& entry : std::filesystem::recursive_directory_iterator(dir)) {
        if (entry.is_regular_file() && entry.path().extension() == ".wav") {
            wavs.push_back(entry.path());
        }
    }
    std::sort(wavs.begin(), wavs.end());

    for (const auto& wav : wavs) {
        const auto rel = std::filesystem::relative(wav, dir).string();
        const auto basename = wav.filename().string();
        std::optional<int> expected;
        int tolerance = 0;
        auto it = expectations.find(rel);
        if (it == expectations.end()) it = expectations.find(basename);
        if (it != expectations.end()) {
            expected = it->second.first;
            tolerance = it->second.second;
        }

        auto r = replayWav(wav, cfg, detector, sm, expected, tolerance);
        if (r.pass) result.passed++;
        else        result.failed++;
        result.total++;
        result.files.push_back(std::move(r));
    }

    if (!cfg.quiet) {
        std::cout << "Replay corpus: " << result.passed << "/" << result.total << " passed\n";
    }
    return result;
}

bool writeJsonOutput(const std::filesystem::path& out_path,
                     const ReplayConfig& cfg,
                     const DirReplayResult& result) {
    nlohmann::json doc;
    doc["config_path"] = cfg.config_path.string();
    doc["profile_path"] = cfg.profile_path.string();
    doc["files"] = nlohmann::json::array();
    for (const auto& r : result.files) {
        nlohmann::json f;
        f["wav"] = r.wav.string();
        f["duration_s"] = r.duration_s;
        f["sample_rate"] = r.sample_rate;
        f["channels"] = r.channels;
        f["expected_triggers"] = r.expected_triggers.has_value() ? nlohmann::json(*r.expected_triggers) : nlohmann::json(nullptr);
        f["observed_triggers"] = r.observed_triggers;
        f["tolerance"] = r.tolerance;
        f["pass"] = r.pass;
        f["triggers"] = nlohmann::json::array();
        for (const auto& t : r.triggers) {
            nlohmann::json tev;
            tev["t_s"] = t.t_s;
            tev["confidence"] = t.confidence;
            tev["state"] = t.state;
            f["triggers"].push_back(tev);
        }
        doc["files"].push_back(f);
    }
    nlohmann::json summary;
    summary["total"] = result.total;
    summary["passed"] = result.passed;
    summary["failed"] = result.failed;
    doc["summary"] = summary;

    std::ofstream out(out_path);
    if (!out.is_open()) return false;
    out << doc.dump(2);
    return true;
}

bool writeJsonOutput(const std::filesystem::path& out_path,
                     const ReplayConfig& cfg,
                     const ReplayResult& single) {
    DirReplayResult wrap;
    wrap.files.push_back(single);
    wrap.total = 1;
    wrap.passed = single.pass ? 1 : 0;
    wrap.failed = single.pass ? 0 : 1;
    return writeJsonOutput(out_path, cfg, wrap);
}
```

  ```python
  envelope = 0.5 * (1 + np.sin(2*np.pi*3*t)) * 0.4   # 3 Hz syllable rate
  ```
