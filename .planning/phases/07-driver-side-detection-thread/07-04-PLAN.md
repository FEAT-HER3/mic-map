// PRESERVE VERBATIM: weak_ptr<State> + alive check at the head (Pitfall 13).

        auto sp = weak.lock();
        if (!sp || !sp->alive.load(std::memory_order_acquire)) return;   // Pitfall 13
        sp->frames_seen.fetch_add(1, std::memory_order_relaxed);

        const bool dropped = ring_ptr->try_push(samples, count);

            // mitigation; the smell of "ring overflowing under load" is
            // surfaced without per-drop spam.
            const uint32_t total = ring_ptr->drops();
            if ((total % kRingDropLogPeriod) == 1) {

            }

        // P7 D-05 step 3: notify the detection thread.
        auto* runner = sp->runner_ptr.load(std::memory_order_acquire);
        if (runner) {

        // P6 RMS log retained behind debug define (D-05 step 4).
        double sumSq = 0.0;
        for (size_t i = 0; i < count; ++i) sumSq += samples[i] * samples[i];
        const float rms = static_cast<float>(std::sqrt(sumSq / std::max<size_t>(count, 1)));
        const uint32_t emitted = sp->rms_logs_emitted.fetch_add(1, std::memory_order_relaxed);

# - Before the setter fires, runner_ptr is nullptr → callback skips NotifyOne.
# - After the setter fires, runner_ptr points to the live DetectionRunner.
