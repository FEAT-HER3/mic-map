    int argc=0;
    LPWSTR* argvW = CommandLineToArgvW(GetCommandLineW(), &argc);
    CliFlags flags = parseCliArgs(argc, argvW);
    LocalFree(argvW);

    if (flags.registerManifest || flags.unregisterManifest) {
        common::Logger::initialize();  // file sink only (D-04)
        vr::EVRInitError e = vr::VRInitError_None;
        vr::VR_Init(&e, vr::VRApplication_Utility);
        if (e != vr::VRInitError_None) {
            MICMAP_LOG_ERROR("CLI VR_Init(Utility) failed: ", vr::VR_GetVRInitErrorAsEnglishDescription(e));
            return 1;
        }
        auto reg = steamvr::createManifestRegistrar();
        int rc = 0;
        if (flags.registerManifest)   rc = (reg->registerApp()   == steamvr::RegisterResult::Success) ? 0 : 1;
        else                          rc = (reg->unregisterApp() == steamvr::RegisterResult::Success) ? 0 : 1;
        vr::VR_Shutdown();
        return rc;
    }

    HANDLE hMutex = CreateMutexW(nullptr, TRUE, L"MicMapSingleInstance");
    if (GetLastError() == ERROR_ALREADY_EXISTS) {
        if (!flags.minimized) {  // D-08: skip focus steal when SteamVR is re-launching

            if (w) { PostMessageW(w, WM_COMMAND, IDM_SHOW, 0); SetForegroundWindow(w); }
        }
        return 0;
    }
    // ... existing D3D/ImGui/tray init ...
    if (flags.minimized) { g_app.minimizedToTray = true; }
    else { ShowWindow(g_app.hwnd, nCmdShow); UpdateWindow(g_app.hwnd); }

    namespace micmap::common {

    CliFlags parseCliArgs(int argc, const wchar_t* const* argv) {
        CliFlags flags{};
        if (!argv) return flags;
        // argv[0] is the exe name — start at index 1.
        for (int i = 1; i < argc; ++i) {
            if (!argv[i]) continue;
            if (std::wcscmp(argv[i], L"--register-vrmanifest") == 0)        flags.registerManifest = true;

        }
        return flags;
    }

    } // namespace micmap::common
