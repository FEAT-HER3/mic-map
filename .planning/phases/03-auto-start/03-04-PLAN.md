<critical_pitfall>
**FORWARD-SLASH MANIFEST PATH KILLS AUTO-LAUNCH SILENTLY** (discovered Plan 03-02 Task 2, 2026-04-23 — see `.planning/phases/03-auto-start/03-02-A2-RESULT.md`).

When `vr::VRApplications()->AddApplicationManifest(absPath, false)` receives a UTF-8 path containing `/` instead of `\`, vrserver fails to extract the parent directory, treats the **entire path as the working directory**, logs `App bigscreen.micmap Working directory <path> is invalid. Skipping`, and silently SKIPS the manifest. No error code is returned to the caller. Auto-launch never fires.

Hard rules enforced by this plan:

1. The wide-char path computed by `resolveManifestAbsolutePath()` MUST end up backslash-canonical. `GetModuleFileNameW` already returns native `\` on Windows; `PathCchRemoveFileSpec` preserves them; the manual append `r += L"\\app.vrmanifest"` uses `\\` deliberately. **Do not** introduce `std::filesystem::path::generic_string()` or any `std::replace(begin, end, '\\', '/')` in the path pipeline.
2. After `wideToUtf8` conversion, before calling `AddApplicationManifest`, the impl MUST guard:
   ```cpp
   if (utf8Path.find('/') != std::string::npos) {
       lastError_ = "manifest path contains forward slash; SteamVR will silently skip. Path: " + utf8Path;
       MICMAP_LOG_ERROR(lastError_);
       return RegisterResult::AddFailed;
   }
   ```
   Treat this as a Rule 2 (correctness) addition during Task 2 — it is not optional.
3. The unit test `test_manifest_registrar` SHOULD include a forward-slash-path regression case that asserts `registerApp` returns `AddFailed` with `lastError` containing `"forward slash"` (Plan 03-04 Task 3 should add this assertion to the existing test if not already present).
</critical_pitfall>

    ```cpp
    /**
     * @file manifest_registrar.cpp
     * @brief SteamVR app.vrmanifest registration (implementation)
     *
     * See manifest_registrar.hpp for contract. Caller owns VR_Init / VR_Shutdown.
     * Log discipline per D-17 / D-18: INFO on state change only; no per-tick logs;
     * log enum NAME (not just number) on errors per RESEARCH Pattern 1.
     */

    #include "micmap/steamvr/manifest_registrar.hpp"
    #include "micmap/common/logger.hpp"

    #include <chrono>
    #include <codecvt>
    #include <locale>
    #include <thread>

    #ifdef MICMAP_HAS_OPENVR
    #include <openvr.h>
    #include <windows.h>
    #include <pathcch.h>
    #endif

    namespace micmap::steamvr {

    namespace {

    constexpr const char* kAppKey        = "bigscreen.micmap";
    constexpr int         kPollIntervalMs = 100;
    constexpr int         kPollMaxAttempts = 20;  // 2000ms ceiling per D-17

    // UTF-16 -> UTF-8 for passing wide paths to OpenVR (which takes const char*).
    std::string wideToUtf8(const std::wstring& w) {
        if (w.empty()) return {};
        int n = WideCharToMultiByte(CP_UTF8, 0, w.c_str(), (int)w.size(), nullptr, 0, nullptr, nullptr);
        std::string out(n, 0);
        WideCharToMultiByte(CP_UTF8, 0, w.c_str(), (int)w.size(), out.data(), n, nullptr, nullptr);
        return out;
    }

    #ifdef MICMAP_HAS_OPENVR
    // Production adapter — forwards to vr::VRApplications() accessor.
    // NO hardcoded "IVRApplications_008" string per RESEARCH Anti-Patterns + A1.
    class VRApplicationsAdapter : public IVRApplicationsSurface {
    public:
        uint32_t AddApplicationManifest(const char* path, bool bTemporary) override {
            auto* apps = vr::VRApplications();
            if (!apps) return 0xFFFFFFFFu;  // VRNotAvailable sentinel
            return (uint32_t)apps->AddApplicationManifest(path, bTemporary);
        }
        bool IsApplicationInstalled(const char* appKey) override {
            auto* apps = vr::VRApplications();
            if (!apps) return false;
            return apps->IsApplicationInstalled(appKey);
        }
        uint32_t SetApplicationAutoLaunch(const char* appKey, bool bAutoLaunch) override {
            auto* apps = vr::VRApplications();
            if (!apps) return 0xFFFFFFFFu;
            return (uint32_t)apps->SetApplicationAutoLaunch(appKey, bAutoLaunch);
        }
        uint32_t RemoveApplicationManifest(const char* path) override {
            auto* apps = vr::VRApplications();
            if (!apps) return 0xFFFFFFFFu;
            return (uint32_t)apps->RemoveApplicationManifest(path);
        }
        const char* GetApplicationsErrorNameFromEnum(uint32_t err) override {
            auto* apps = vr::VRApplications();
            if (!apps) return "VRNotAvailable";
            return apps->GetApplicationsErrorNameFromEnum((vr::EVRApplicationError)err);
        }
    };

    // Long-path-safe manifest path resolution (D-21, RESEARCH Pitfall 9).
    std::wstring resolveManifestAbsolutePath() {
        WCHAR buf[32768];  // long-path safe
        DWORD n = GetModuleFileNameW(nullptr, buf, _countof(buf));
        if (n == 0 || n == _countof(buf)) {
            MICMAP_LOG_ERROR("GetModuleFileNameW failed or truncated");
            return {};
        }
        if (FAILED(PathCchRemoveFileSpec(buf, _countof(buf)))) {
            MICMAP_LOG_ERROR("PathCchRemoveFileSpec failed");
            return {};
        }
        std::wstring r = buf;
        r += L"\\app.vrmanifest";
        return r;
    }
    #endif // MICMAP_HAS_OPENVR

    // Shared impl, agnostic of production-vs-test surface.
    class ManifestRegistrarImpl : public IManifestRegistrar {
    public:
        ManifestRegistrarImpl(IVRApplicationsSurface& surface, std::string appKey, std::wstring manifestAbsPath)
            : surface_(surface), appKey_(std::move(appKey)), manifestAbsPath_(std::move(manifestAbsPath)) {}

        RegisterResult registerApp() override {
            const std::string utf8Path = wideToUtf8(manifestAbsPath_);
            if (utf8Path.empty()) {
                lastError_ = "manifest path empty or invalid";
                MICMAP_LOG_ERROR(lastError_);
                return RegisterResult::AddFailed;
            }

            uint32_t err = surface_.AddApplicationManifest(utf8Path.c_str(), /*bTemporary=*/false);
            if (err != 0u) {
                lastError_ = std::string("AddApplicationManifest: ") + surface_.GetApplicationsErrorNameFromEnum(err);
                MICMAP_LOG_WARNING(lastError_);
                return RegisterResult::AddFailed;
            }

            MICMAP_LOG_INFO("polling for manifest install");  // D-17 single entry log
            bool installed = false;
            for (int i = 0; i < kPollMaxAttempts; ++i) {
                if (surface_.IsApplicationInstalled(appKey_.c_str())) {
                    installed = true;
                    MICMAP_LOG_INFO("manifest ready after ", (i + 1) * kPollIntervalMs, "ms");
                    break;
                }
                std::this_thread::sleep_for(std::chrono::milliseconds(kPollIntervalMs));
            }
            if (!installed) {
                lastError_ = "IsApplicationInstalled timeout after 2000ms";
                MICMAP_LOG_WARNING("timeout after 2000ms");
                return RegisterResult::PollTimeout;
            }

            err = surface_.SetApplicationAutoLaunch(appKey_.c_str(), /*bAutoLaunch=*/true);
            if (err != 0u) {
                lastError_ = std::string("SetApplicationAutoLaunch: ") + surface_.GetApplicationsErrorNameFromEnum(err);
                MICMAP_LOG_WARNING(lastError_);
                return RegisterResult::AutoLaunchFailed;
            }
            MICMAP_LOG_INFO("manifest registered + auto-launch enabled: ", appKey_);
            lastError_.clear();
            return RegisterResult::Success;
        }

        RegisterResult unregisterApp() override {
            const std::string utf8Path = wideToUtf8(manifestAbsPath_);
            if (utf8Path.empty()) {
                lastError_ = "manifest path empty or invalid";
                return RegisterResult::RemoveFailed;
            }
            uint32_t err = surface_.RemoveApplicationManifest(utf8Path.c_str());
            if (err != 0u) {
                lastError_ = std::string("RemoveApplicationManifest: ") + surface_.GetApplicationsErrorNameFromEnum(err);
                MICMAP_LOG_WARNING(lastError_);
                return RegisterResult::RemoveFailed;
            }
            MICMAP_LOG_INFO("manifest removed: ", appKey_);
            lastError_.clear();
            return RegisterResult::Success;
        }

        RegisterResult ensureRegistered() override {
            if (surface_.IsApplicationInstalled(appKey_.c_str())) {
                // D-18: log on state change ONLY — no output here.
                return RegisterResult::Success;
            }
            return registerApp();  // logs INFO on transition per D-18
        }

        std::string getLastError() const override { return lastError_; }

    private:
        IVRApplicationsSurface& surface_;
        std::string             appKey_;
        std::wstring            manifestAbsPath_;
        std::string             lastError_;
    };

    // Stub for MICMAP_HAS_OPENVR-undefined builds. Returns VRNotAvailable on all ops.
    class StubManifestRegistrar : public IManifestRegistrar {
    public:
        RegisterResult registerApp()      override { return RegisterResult::VRNotAvailable; }
        RegisterResult unregisterApp()    override { return RegisterResult::VRNotAvailable; }
        RegisterResult ensureRegistered() override { return RegisterResult::VRNotAvailable; }
        std::string    getLastError() const override { return "OpenVR not available at build time"; }
    };

    } // anon namespace

    std::unique_ptr<IManifestRegistrar> createManifestRegistrar() {
    #ifdef MICMAP_HAS_OPENVR
        static VRApplicationsAdapter s_adapter;  // stateless; safe as static
        std::wstring manifestPath = resolveManifestAbsolutePath();
        if (manifestPath.empty()) {
            MICMAP_LOG_ERROR("createManifestRegistrar: manifest path resolution failed; returning stub");
            return std::make_unique<StubManifestRegistrar>();
        }
        return std::unique_ptr<IManifestRegistrar>(
            new ManifestRegistrarImpl(s_adapter, std::string(kAppKey), std::move(manifestPath)));
    #else
        MICMAP_LOG_WARNING("OpenVR not available — using stub manifest registrar");
        return std::make_unique<StubManifestRegistrar>();
    #endif
    }

    std::unique_ptr<IManifestRegistrar> createManifestRegistrarForTesting(
        IVRApplicationsSurface& surface,
        std::string appKey,
        std::wstring manifestAbsPath) {
        return std::unique_ptr<IManifestRegistrar>(
            new ManifestRegistrarImpl(surface, std::move(appKey), std::move(manifestAbsPath)));
    }

    } // namespace micmap::steamvr
    ```
