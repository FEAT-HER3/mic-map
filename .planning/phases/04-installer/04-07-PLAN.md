It runs four steps via Inno's `Exec()` with explicit `ResultCode` inspection (Technique B per 04-RESEARCH.md §Open Question 9):

1. `vrpathreg.exe removedriver "{app}"` — unconditional, exit code ignored (removedriver on unregistered path is a documented no-op; INST-03 + Pitfall 3).
2. `vrpathreg.exe adddriver "{app}"` — expected rc=0; failure recorded, does not abort (INST-03).
3. `{app}\bin\micmap.exe --register-vrmanifest` — expected rc=0 per Phase 3 D-03 exit contract; failure recorded (INST-04).
4. `{app}\bin\micmap.exe --patch-bindings` — expected rc=0 per Plan 02 CLI contract; failure recorded (INST-08 orchestration; the binding lib itself is Plans 01-02).

Every `vrpathreg.exe` call is gated by `VrpathregExists` (Pitfall 10: user may have uninstalled Steam independently). Failures aggregate into a `TStringList` and are surfaced to the user via a single end-of-install `MsgBox` directing them at `%APPDATA%\MicMap\micmap.log` for diagnostics.

Closes **INST-03** (removedriver-before-adddriver unconditional; FileExists gate) and **INST-04** (--register-vrmanifest post-install invocation). Wires **INST-08** at the installer level (the CLI entry points are delivered by Plans 01-02).

Purpose: Technique B replaces the naive `[Run]` block entirely because `[Run]` silently ignores non-zero exit codes (Pitfall 17), making a failed `--register-vrmanifest` or `--patch-bindings` invisible to the user and the installer log. `Exec()` + `ResultCode` makes the failure observable.

```pascal
function GetVrpathreg(Param: String): String;
begin
  // g_SteamVRDir is populated by Plan 05 InitializeSetup.
  Result := g_SteamVRDir + '\bin\win64\vrpathreg.exe';
end;

function VrpathregExists(): Boolean;
begin
  Result := FileExists(GetVrpathreg(''));
end;

procedure CurStepChanged(CurStep: TSetupStep);
var
  ResultCode: Integer;
  Failed: TStringList;
  AppDir: String;
  MicMapExe: String;
begin
  if CurStep <> ssPostInstall then
    Exit;

  AppDir := ExpandConstant('{app}');
  MicMapExe := AppDir + '\bin\micmap.exe';
  Failed := TStringList.Create;
  try
    // Step 1: vrpathreg removedriver (Pitfall 3 — unconditional, always before adddriver).
    // rc is INTENTIONALLY ignored — removedriver on an unregistered path returns 0,
    // and any other failure is still fine because Step 2 (adddriver) is what matters.
    if VrpathregExists() then
      Exec(GetVrpathreg(''),
           'removedriver "' + AppDir + '"',
           '',
           SW_HIDE, ewWaitUntilTerminated,
           ResultCode);

    // Step 2: vrpathreg adddriver (Pitfall 15 — pass "{app}" which IS the manifest dir,
    // NOT {app}\bin\win64 which is the DLL dir).
    if VrpathregExists() then
    begin
      if (not Exec(GetVrpathreg(''),
                   'adddriver "' + AppDir + '"',
                   '',
                   SW_HIDE, ewWaitUntilTerminated,
                   ResultCode))
         or (ResultCode <> 0) then
        Failed.Add('vrpathreg adddriver (rc=' + IntToStr(ResultCode) + ')');
    end;

    // Step 3: --register-vrmanifest (Phase 3 D-03 exit contract: 0=success, 1=failure).
    if (not Exec(MicMapExe,
                 '--register-vrmanifest',
                 '',
                 SW_HIDE, ewWaitUntilTerminated,
                 ResultCode))
       or (ResultCode <> 0) then
      Failed.Add('micmap.exe --register-vrmanifest (rc=' + IntToStr(ResultCode) + ')');

    // Step 4: --patch-bindings (same exit contract per Plan 02 CLI).
    // Failure is NOT catastrophic: driver-side patcher (driver/src/bindings_patcher.cpp,
    // now micmap::bindings after Plan 01 lift) re-runs on every driver init as a
    // defensive fallback. App-side patch is eager/pre-emptive.
    if (not Exec(MicMapExe,
                 '--patch-bindings',
                 '',
                 SW_HIDE, ewWaitUntilTerminated,
                 ResultCode))
       or (ResultCode <> 0) then
      Failed.Add('micmap.exe --patch-bindings (rc=' + IntToStr(ResultCode) + ')');

    if Failed.Count > 0 then
      MsgBox('MicMap installed, but some post-install steps failed:'#13#10#13#10 +
             Failed.Text +
             #13#10'MicMap will still work for basic dashboard toggling. ' +
             'If problems persist, see %APPDATA%\MicMap\micmap.log and re-run the installer.',
             mbInformation, MB_OK);
  finally
    Failed.Free;
  end;
end;
```
