// driver/src/audio_worker.hpp
#pragma once

#include <atomic>
#include <condition_variable>
#include <memory>
#include <mutex>
#include <thread>

namespace micmap::audio { class IAudioCapture; }   // forward-decl, no include in header

namespace micmap::driver {

class AudioWorker {
public:
    AudioWorker();
    ~AudioWorker();

    bool Start();   // spawns thread_; returns true if joinable
    void Stop();    // idempotent; called by ~AudioWorker

    bool IsRunning() const { return running_; }

    // Pitfall 13 alive-flag mitigation lives on this State, captured by
    // weak_ptr inside the audio callback so callbacks bail when alive=false.
    struct State {
        std::atomic<bool>     alive{true};
        std::atomic<uint32_t> rms_logs_emitted{0};
        std::atomic<uint32_t> frames_seen{0};
    };

    // Test-only accessor (used by tests/driver/audio_worker_lifecycle_headless.cpp
    // case 3 if the planner chooses to enable the alive-before-shutdown assertion;
    // otherwise unused — does NOT widen the production surface).
    std::shared_ptr<State> state_for_test() const { return state_; }

private:
    static void ThreadEntry(AudioWorker* self);
    void RunWorker();

    std::shared_ptr<State>          state_;
    std::unique_ptr<micmap::audio::IAudioCapture> capture_;  // constructed/destructed on worker
    std::thread                     thread_;
    std::mutex                      mu_;
    std::condition_variable         cv_;
    std::atomic<bool>               shutdown_{false};
    std::atomic<bool>               running_{false};
    std::atomic<bool>               thread_finished_{false};   // for 2 s watchdog poll
};

} // namespace micmap::driver

RMS callback with weak_ptr alive-flag (Pitfall 13, D-16):
```cpp
constexpr uint32_t kRmsBudget = 100;   // ~1 s @ 10 ms WASAPI shared-mode period (D-08)

std::weak_ptr<State> weak = state_;
capture_->setAudioCallback(
    [weak](const float* samples, size_t count) {
        auto sp = weak.lock();
        if (!sp || !sp->alive.load(std::memory_order_acquire)) return;
        sp->frames_seen.fetch_add(1, std::memory_order_relaxed);
        double sumSq = 0.0;
        for (size_t i = 0; i < count; ++i) sumSq += samples[i] * samples[i];
        float rms = static_cast<float>(std::sqrt(sumSq / std::max<size_t>(count, 1)));
        uint32_t emitted = sp->rms_logs_emitted.fetch_add(1, std::memory_order_relaxed);
        if (emitted < kRmsBudget) {
            DriverLog("MicMap audio: rms[%u]=%.6f\n", emitted, rms);
        }
        // After budget: continue draining frames so WASAPI buffer never overflows,
        // but skip DriverLog writes (D-08 — log-flood mitigation, T6 / Pitfall 6 carry).
    });
```

Stop() with 2 s watchdog (D-13 — strengthen HttpServer::Stop analog from http_server.cpp:102-122):
```cpp
void AudioWorker::Stop() {
    if (!thread_.joinable()) return;
    if (state_) state_->alive.store(false, std::memory_order_release);
    {
        std::lock_guard<std::mutex> lk(mu_);
        shutdown_.store(true, std::memory_order_release);
    }
    cv_.notify_all();

    // 2 s watchdog matching v1.5 VREvent_Quit precedent. Poll thread_finished_
    // every 25 ms; on overrun, log + detach (last-resort to avoid blocking
    // vrserver.exe shutdown — T3 mitigation).
    using namespace std::chrono;
    const auto deadline = steady_clock::now() + seconds(2);
    while (!thread_finished_.load(std::memory_order_acquire) &&
           steady_clock::now() < deadline) {
        std::this_thread::sleep_for(milliseconds(25));
    }
    if (thread_finished_.load(std::memory_order_acquire)) {
        thread_.join();
    } else {
        DriverLog("MicMap: audio worker did not exit within 2 s watchdog — detaching (T3 mitigation)\n");
        thread_.detach();
    }
    running_.store(false, std::memory_order_release);
}

AudioWorker::~AudioWorker() { Stop(); }
```
