```cpp
struct StateMachineConfig {
    std::chrono::milliseconds minDetectionDuration{100};  // D-04: was 500; min-press floor on the app side
    std::chrono::milliseconds minReleaseDuration{80};     // NEW per D-10: symmetric debounce on confidence drop
    std::chrono::milliseconds cooldownDuration{200};      // D-11: semantic shift — NOW post-release only
    float detectionThreshold{0.7f};
};

enum class State {
    Idle, Training, Detecting, Triggered,
    Releasing,     // NEW — debounces the UP edge; exits to Cooldown after minReleaseDuration
    Cooldown
};

enum class PressEdge { Down, Up };        // NEW

using TriggerCallback = std::function<void(PressEdge)>;   // CHANGED from std::function<void()>

// stateToString() gains a "Releasing" case.
```

### src/core/src/state_machine.cpp (flow)

```
Idle
  → Detecting       (confidence >= threshold)
Detecting
  → Idle            (confidence < threshold && timeInState < minDetectionDuration)
  → Triggered       (timeInState >= minDetectionDuration)
                    [emit triggerCallback_(PressEdge::Down)]
Triggered
  → Releasing       (confidence < threshold)
Releasing
  → Triggered       (confidence >= threshold; UP was never emitted)
  → Cooldown        (timeInState >= minReleaseDuration)
                    [emit triggerCallback_(PressEdge::Up)]
Cooldown
  → Idle            (timeInState >= cooldownDuration)
```
