```cpp
void MicMapApp::pollDriverHealth() {
    if (!driverClient) return;

    auto now = std::chrono::steady_clock::now();

    bool isMinimized = ::IsIconic(hwnd) != FALSE;

    // /health - 1 Hz both visible and minimized.
    // P8 Pitfall 6 (per checker context_compliance fix): differentiate
    // ConnectResult::NotFound from Timeout. Pitfall 6 mitigation in 08-01
    // exists specifically so the indicator does NOT flip red on transient
    // Read/Write timeouts (driver up but slow); only ECONNREFUSED (NotFound)
    // is the canonical "driver down" signal per HEALTH-01. Treating Timeout
    // identically to NotFound here would re-introduce the very flicker
    // 08-01 Task 1 was designed to eliminate.
    if (now - lastHealthPoll >= std::chrono::milliseconds(1000)) {
        lastHealthPoll = now;
        // Try the cheap path first: if already connected and isConnected()
        // returns true, treat as Connected. Otherwise reissue connect() so
        // we get a fresh ConnectResult to discriminate.
        steamvr::ConnectResult r;
        if (driverClient->isConnected()) {
            r = steamvr::ConnectResult::Connected;
        } else {
            r = driverClient->connect();
        }
        switch (r) {
            case steamvr::ConnectResult::Connected:
                driverLoadedIndicator.store(true);
                steamvrRunningIndicator.store(true);
                break;
            case steamvr::ConnectResult::NotFound:
                // ECONNREFUSED on every port - driver not loaded. Flip RED.
                MICMAP_LOG_DEBUG("Driver poll: NotFound (ECONNREFUSED)");
                driverLoadedIndicator.store(false);
                steamvrRunningIndicator.store(false);
                break;
            case steamvr::ConnectResult::Timeout:
                // Driver MAY be alive but slow on /health Read/Write. Per
                // Pitfall 6 fix in 08-01 we DO NOT flip the indicator on
                // Timeout - leave the prior value intact. This avoids
                // false-red flicker during transient slowdowns (vrserver
                // GC pause, contended thread pool, etc.).
                MICMAP_LOG_DEBUG("Driver poll: Timeout (keeping prior indicator state)");
                break;
            case steamvr::ConnectResult::OtherError:
            default:
                // Documented choice (per checker fix_hint): treat OtherError
                // as red (same as NotFound). Rationale: unclassified failures
                // (DNS, malformed URL) are programmer-error or environment
                // breakage; surfacing them as "driver down" to the user is
                // accurate and the actionable UX is the same as NotFound
                // (re-launch / re-install).
                MICMAP_LOG_DEBUG("Driver poll: OtherError (treating as red)");
                driverLoadedIndicator.store(false);
                steamvrRunningIndicator.store(false);
                break;
        }
    }

    if (!driverLoadedIndicator.load()) return;   // skip subsequent polls if down

    // /state — 2 Hz visible / 0.5 Hz minimized.
    auto stateInterval = isMinimized ? std::chrono::milliseconds(2000)
                                     : std::chrono::milliseconds(500);
    if (now - lastStatePoll >= stateInterval) {
        lastStatePoll = now;
        auto state = driverClient->getState();
        if (state.has_value()) {
            std::lock_guard<std::mutex> lk(healthMu);
            detectionStateStr = state->detection_state;
            lastTriggerAt     = state->last_trigger_at;
            lastError         = state->last_error;
            audioDeviceState  = state->audio_device_state;
        }
    }

    // /telemetry/level — 5 Hz visible / 0.5 Hz minimized.
    auto levelInterval = isMinimized ? std::chrono::milliseconds(2000)
                                     : std::chrono::milliseconds(200);
    if (now - lastLevelPoll >= levelInterval) {
        lastLevelPoll = now;
        auto lvl = driverClient->getTelemetryLevel();
        if (lvl.has_value()) {
            driverLevelDbfs.store(lvl->dbfs);
            driverLevelRmsNormalized.store(lvl->rms_normalized);
        }
    }
}
```
